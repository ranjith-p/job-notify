# LinkedIn job alert search configuration
#
# Edit this file to customize what linkedin_alert.py searches for and
# filters. After editing, commit and push — the next run picks it up
# automatically. Same format as search_config.txt (used by job_alert.py /
# Adzuna) — no ADZUNA_COUNTRY section here, not applicable to LinkedIn.
# =============================================================================

# ---------------------------------------------------------------------------
# KEYWORDS — job title/description phrases to search for (one per line).
# Each is queried separately against LinkedIn's public search; results are
# merged and deduplicated by job ID.
# ---------------------------------------------------------------------------
[KEYWORDS]
data scientist
data science
machine learning engineer
ML engineer
applied scientist
research scientist
data analyst


# ---------------------------------------------------------------------------
# EXCLUDE_TITLES — if a job's title contains any of these (as a whole
# word/phrase, case-insensitive), it's skipped entirely. One per line.
# ---------------------------------------------------------------------------
[EXCLUDE_TITLES]
praktikum
praktikant
praktikantin
werkstudent
werkstudentin
intern
internship
working student
duales studium
dual studium
dual study
software engineer
data engineer


# ---------------------------------------------------------------------------
# EXCLUDE_GERMAN_LEVELS — job is skipped if the LLM determines German is a
# MANDATORY requirement at one of these levels. One per line. Common
# values: b1, b2, c1, c2, fluent, native. Jobs where German is mentioned
# as optional/nice-to-have are never excluded, regardless of this list.
# ---------------------------------------------------------------------------
[EXCLUDE_GERMAN_LEVELS]
c1
c2
fluent
native


# ---------------------------------------------------------------------------
# LOCATIONS — passed directly to LinkedIn's location search field. One per
# line. "Germany" searches nationwide; add specific cities too if you want
# — since only page 1 (10 results) is fetched per keyword/location combo,
# a city-specific search can catch listings a broad nationwide query's
# top-10 cutoff might otherwise miss.
# ---------------------------------------------------------------------------
[LOCATIONS]
Berlin, Germany
Germany


# ---------------------------------------------------------------------------
# MIN_MATCH_SCORE — jobs scored below this (out of 10) by the LLM are
# skipped. Single number, 1-10. Jobs where scoring itself failed are
# still sent regardless of this setting.
# ---------------------------------------------------------------------------
[MIN_MATCH_SCORE]
4


# ---------------------------------------------------------------------------
# MAX_POSTING_AGE_DAYS — a job is skipped (but still recorded as seen, so
# it's not re-checked every run) if its LinkedIn-listed posting date is
# older than this many days. LinkedIn's public search has no reliable
# freshness filter, so a low-traffic keyword/location combo can surface a
# genuinely old posting for the first time just because nothing newer has
# pushed it off page 1 yet — this catches that case specifically.
# ---------------------------------------------------------------------------
[MAX_POSTING_AGE_DAYS]
3


# ---------------------------------------------------------------------------
# PAGES_PER_QUERY — how many results pages (10 jobs each) to fetch per
# keyword/location combo, per run. Page 1 only means this bot can never see
# past the top 10 results for a given query - if you're missing postings
# that show up when you search LinkedIn yourself, raising this is the fix.
# Each extra page is one more request per combo (7 keywords x 2 locations
# x this number, per run) - weigh against request volume / IP risk, see
# the "About the LinkedIn source" section in README.md.
# ---------------------------------------------------------------------------
[PAGES_PER_QUERY]
2


# ---------------------------------------------------------------------------
# Groq model and search parameters
#
# NOTE: openai/gpt-oss-20b's free tier is 30 requests/min, 1,000 requests/
# DAY, 8,000 tokens/min - and that budget is SHARED with job_alert.py
# (Adzuna), since both scripts use the same GROQ_API_KEY. Running this
# workflow frequently adds to that shared daily budget fast, especially
# alongside Adzuna's own usage. If you see "Groq quota exhausted" warnings
# in the Action logs, the fix is a less frequent cron-job.org schedule for
# one or both workflows, not a bigger GROQ_CALL_PACING_SECONDS - that only
# paces requests within a single run, it can't stretch a 1,000/day cap
# that both scripts draw from.
# ---------------------------------------------------------------------------
[GROQ_MODEL]
openai/gpt-oss-20b


[GROQ_CALL_PACING_SECONDS]
15


[MAX_JOBS_SCORED_PER_RUN]
60

[MAX_DESCRIPTION_CHARS]
2000
